# Next.js Architecture

Next.js is a React framework that enables production-grade full-stack applications. This guide covers organizing large-scale Next.js applications with App Router, server components, and modern best practices.

## Key Principles

- **File-Based Routing:** Automatic routing from file structure
- **Server & Client Components:** Choose where to render based on needs
- **API Routes/Handlers:** Built-in backend with `/api` routes
- **Middleware:** Request-level logic with Edge Runtime support
- **Data Fetching:** Server-side fetching and revalidation
- **Optimization:** Automatic image, font, and script optimization
- **Full-Stack:** Frontend and backend in one codebase

### Dynamic Route (Server Component)

```typescript

import { notFound } from 'next/navigation';
import { UserForm } from '@/components/users/UserForm';
import { usersService } from '@/services/users.service';

export async function generateMetadata({
  params,
}: {
  params: { id: string };
}) {
  const user = await usersService.getUserById(params.id);
  if (!user) return { title: 'User not found' };
  return { title: `Edit ${user.name}` };
}

export async function generateStaticParams() {
  const users = await usersService.getUsers({ limit: 100 });
  return users.map((user) => ({
    id: user.id,
  }));
}

export default async function UserDetailPage({
  params,
}: {
  params: { id: string };
}) {
  const user = await usersService.getUserById(params.id);

  if (!user) {
    notFound();
  }

  return <UserForm initialUser={user} />;
}
```

## Client Components

### Interactive Component with Hooks

```typescript
// app/(dashboard)/users/components/UserForm.tsx
'use client';

import { FormEvent, useState } from 'react';
import { User } from '@/types/models.types';

interface UserFormProps {
  initialUser?: User;
}

export function UserForm({ initialUser }: UserFormProps) {
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  async function handleSubmit(e: FormEvent<HTMLFormElement>) {
    e.preventDefault();
    setLoading(true);
    setError(null);

    try {
      const formData = new FormData(e.currentTarget);
      const data = Object.fromEntries(formData);

      const response = await fetch(
        initialUser ? `/api/users/${initialUser.id}` : '/api/users',
        {
          method: initialUser ? 'PUT' : 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(data),
        }
      );

      if (!response.ok) throw new Error('Failed to save');

      // Revalidate and redirect
      window.location.href = '/users';
    } catch (err) {
      setError(err instanceof Error ? err.message : 'An error occurred');
    } finally {
      setLoading(false);
    }
  }

  return (
    <form onSubmit={handleSubmit}>
      <input
        type="text"
        name="name"
        defaultValue={initialUser?.name}
        required
      />
      <input
        type="email"
        name="email"
        defaultValue={initialUser?.email}
        required
      />
      <button type="submit" disabled={loading}>
        {loading ? 'Saving...' : 'Save'}
      </button>
      {error && <div className="error">{error}</div>}
    </form>
  );
}
```

## API Routes (Backend)

### GET/POST Handler

```typescript
// app/api/users/route.ts
import { NextRequest, NextResponse } from 'next/server';
import { usersService } from '@/services/users.service';
import { validateCreateUser } from '@/lib/validators';

export async function GET(request: NextRequest) {
  try {
    const { searchParams } = new URL(request.url);
    const page = Number(searchParams.get('page')) || 1;
    const limit = Number(searchParams.get('limit')) || 10;

    const users = await usersService.getUsers({ page, limit });

    return NextResponse.json({
      success: true,
      data: users,
    });
  } catch (error) {
    return NextResponse.json(
      { success: false, error: 'Failed to fetch users' },
      { status: 500 }
    );
  }
}

export async function POST(request: NextRequest) {
  try {
    const body = await request.json();

    // Validate input
    const validation = validateCreateUser(body);
    if (!validation.valid) {
      return NextResponse.json(
        { success: false, errors: validation.errors },
        { status: 400 }
      );
    }

    const user = await usersService.createUser(body);

    return NextResponse.json(
      { success: true, data: user },
      { status: 201 }
    );
  } catch (error) {
    return NextResponse.json(
      { success: false, error: 'Failed to create user' },
      { status: 500 }
    );
  }
}
```

### Dynamic Route Handler

```typescript
// app/api/users/[id]/route.ts
import { NextRequest, NextResponse } from 'next/server';
import { usersService } from '@/services/users.service';

export async function GET(
  request: NextRequest,
  { params }: { params: { id: string } }
) {
  try {
    const user = await usersService.getUserById(params.id);

    if (!user) {
      return NextResponse.json(
        { success: false, error: 'User not found' },
        { status: 404 }
      );
    }

    return NextResponse.json({ success: true, data: user });
  } catch (error) {
    return NextResponse.json(
      { success: false, error: 'Failed to fetch user' },
      { status: 500 }
    );
  }
}

export async function PUT(
  request: NextRequest,
  { params }: { params: { id: string } }
) {
  try {
    const body = await request.json();
    const user = await usersService.updateUser(params.id, body);

    return NextResponse.json({ success: true, data: user });
  } catch (error) {
    return NextResponse.json(
      { success: false, error: 'Failed to update user' },
      { status: 500 }
    );
  }
}

export async function DELETE(
  request: NextRequest,
  { params }: { params: { id: string } }
) {
  try {
    await usersService.deleteUser(params.id);

    return NextResponse.json({
      success: true,
      message: 'User deleted',
    });
  } catch (error) {
    return NextResponse.json(
      { success: false, error: 'Failed to delete user' },
      { status: 500 }
    );
  }
}
```

## Middleware

### Authentication Middleware

```typescript
// middleware.ts
import { NextRequest, NextResponse } from 'next/server';
import { verifyAuth } from '@/lib/auth/jwt';

export async function middleware(request: NextRequest) {
  const token = request.cookies.get('auth-token')?.value;

  // Public routes
  if (request.nextUrl.pathname.startsWith('/login') ||
      request.nextUrl.pathname.startsWith('/signup')) {
    return NextResponse.next();
  }

  // Protected routes
  if (request.nextUrl.pathname.startsWith('/dashboard')) {
    if (!token) {
      return NextResponse.redirect(new URL('/login', request.url));
    }

    try {
      const payload = await verifyAuth(token);
      const response = NextResponse.next();
      response.headers.set('x-user-id', payload.id);
      return response;
    } catch (error) {
      return NextResponse.redirect(new URL('/login', request.url));
    }
  }

  return NextResponse.next();
}

export const config = {
  matcher: [
    '/((?!_next/static|_next/image|favicon.ico).*)',
  ],
};
```

## Services

```typescript
// services/users.service.ts
import { db } from '@/lib/db/client';
import { CreateUserDTO, UpdateUserDTO } from '@/types/dto.types';
import { User } from '@/types/models.types';

class UsersService {
  async getUsers(options: { page: number; limit: number }): Promise<User[]> {
    return db.user.findMany({
      skip: (options.page - 1) * options.limit,
      take: options.limit,
      orderBy: { createdAt: 'desc' },
    });
  }

  async getUserById(id: string): Promise<User | null> {
    return db.user.findUnique({
      where: { id },
    });
  }

  async createUser(data: CreateUserDTO): Promise<User> {
    const existingUser = await db.user.findUnique({
      where: { email: data.email },
    });

    if (existingUser) {
      throw new Error('Email already exists');
    }

    return db.user.create({
      data,
    });
  }

  async updateUser(id: string, data: UpdateUserDTO): Promise<User> {
    return db.user.update({
      where: { id },
      data,
    });
  }

  async deleteUser(id: string): Promise<void> {
    await db.user.delete({
      where: { id },
    });
  }
}

export const usersService = new UsersService();
```

## Best Practices

- **Server vs Client Components:** Use server components by default, client when needed
- **Data Fetching:** Fetch data in server components (no useEffect waterfalls)
- **Revalidation:** Use `revalidatePath()` and `revalidateTag()` for ISR
- **Error Handling:** Implement error.tsx boundaries
- **Security:** Never expose secrets in client-side code
- **Environment Variables:** Use `.env.local` for secrets
- **Middleware:** Use for authentication and request transformation
- **API Routes:** Build minimal backend, delegate heavy logic to services
- **Caching:** Leverage Next.js caching strategies
- **Testing:** Test API routes and services independently

## File Naming Conventions

```
Pages:              page.tsx              (always this name for pages)
Layouts:            layout.tsx            (always this name for layouts)
Error Boundaries:   error.tsx             (always this name)
Not Found:          not-found.tsx         (always this name)
Middleware:         middleware.ts         (always at root)
API Routes:         route.ts              (always this name for API)
Components:         Component.tsx         (PascalCase)
Custom Hooks:       useName.ts            (camelCase)
Services:           name.service.ts       (camelCase)
Types:              name.types.ts         (camelCase)
Utilities:          name.ts               (camelCase)
```
